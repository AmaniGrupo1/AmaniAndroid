# Setup Commands - APK Security Audit
# Generated: 2026-04-18
# Environment: Docker container (apk-auditor)

# ============================================
# DOCKER BUILD COMMANDS
# ============================================

# Build Docker image
docker build -t apk-auditor -f Dockerfile.audit .

# Run analysis container
docker run --rm \
  -v "/home/ivan/AndroidStudioProjects/AmaniAndroid:/analysis" \
  -v "/home/ivan/AndroidStudioProjects/AmaniAndroid/output:/output" \
  apk-auditor \
  bash -c "cd /analysis && bash -x audit.sh"

# ============================================
# APK RECOGNITION - PASO 1
# ============================================

# Check APK location
ls -la /analysis/analysis/*.apk

# File type detection
file /analysis/analysis/app-debug.apk

# List APK contents
unzip -l /analysis/analysis/app-debug.apk | head -80

# APK metadata
aapt dump badging /analysis/analysis/app-debug.apk

# Signature verification
apksigner verify --verbose /analysis/analysis/app-debug.apk

# APK hash
sha256sum /analysis/analysis/app-debug.apk

# ============================================
# DECOMPILATION - PASO 2
# ============================================

# Decompile with apktool
apktool d -f /analysis/analysis/app-debug.apk -o /analysis/output/apktool_output/

# Decompile with jadx
jadx -d /analysis/output/jadx_output --show-bad-code /analysis/analysis/app-debug.apk

# Extract DEX files
unzip -j -o /analysis/analysis/app-debug.apk '*.dex' -d /analysis/output/dex_output/

# Extract native libraries
unzip -j -o /analysis/analysis/app-debug.apk 'lib/*' -d /analysis/output/native_libs/

# ============================================
# MANIFEST ANALYSIS - PASO 3
# ============================================

# Debuggable check
grep -A 3 android:debuggable /analysis/output/apktool_output/AndroidManifest.xml

# Backup enabled check
grep -A 3 android:allowBackup /analysis/output/apktool_output/AndroidManifest.xml

# Cleartext traffic check
grep -A 3 android:usesCleartextTraffic /analysis/output/apktool_output/AndroidManifest.xml

# Network security config
cat /analysis/output/apktool_output/res/xml/network_security_config.xml 2>/dev/null || echo "Not found"

# Permissions list
grep 'uses-permission' /analysis/output/apktool_output/AndroidManifest.xml

# Exported components
grep -E 'android:exported="true"' /analysis/output/apktool_output/AndroidManifest.xml

# Deep links
grep -B 2 -A 5 'intent-filter' /analysis/output/apktool_output/AndroidManifest.xml | grep -A 3 'data android:scheme'

# Content providers
grep -B 2 -A 8 'provider' /analysis/output/apktool_output/AndroidManifest.xml

# ============================================
# SECRET SEARCH - PASO 4
# ============================================

# API keys and secrets
grep -rn -E "(api_key|apikey|secret|password|token|Bearer|Basic |private_key|AWS|firebase)" \
  --include="*.kt" --include="*.java" --include="*.xml" \
  /analysis/output/jadx_output/ 2>/dev/null | head -60

# URLs extraction
grep -rn -E "https?://[a-zA-Z0-9./_-]+" \
  --include="*.kt" --include="*.java" \
  /analysis/output/jadx_output/ 2>/dev/null | sort -u | head -60

# Strings from APK
strings /analysis/analysis/app-debug.apk | grep -E "(http|https)://" | sort -u | head -50
strings /analysis/analysis/app-debug.apk | grep -E "[A-Za-z0-9+/]{40,}={0,2}" | sort -u | head -30
strings /analysis/analysis/app-debug.apk | grep -iE "(firebase|google-services|api_key)" | head -30

# ============================================
# KOTLIN/COMPOSE ANALYSIS - PASO 5
# ============================================

# Companion objects with secrets
grep -rn "companion object" /analysis/output/jadx_output/ -A 10 | grep -E "(key|secret|token|url|password)"

# BuildConfig leaks
grep -rn "BuildConfig\." /analysis/output/jadx_output/ | grep -E "(API|KEY|SECRET|TOKEN|URL|DEBUG)"

# StateFlow
grep -rn "MutableStateFlow\|MutableSharedFlow" /analysis/output/jadx_output/ -A 5

# WebViews
grep -rn "WebView\|setJavaScriptEnabled\|addJavascriptInterface" /analysis/output/jadx_output/

# Cryptography
grep -rn "Cipher\|SecretKey\|KeyGenerator\|MessageDigest\|KeyStore" /analysis/output/jadx_output/

# Room/DataStore
grep -rn "@Database\|@Entity\|DataStore\|PreferenceDataStore" /analysis/output/jadx_output/ -A 5

# SharedPreferences
grep -rn "SharedPreferences\|getSharedPreferences" /analysis/output/jadx_output/

# EncryptedSharedPreferences
grep -rn "EncryptedSharedPreferences\|EncryptedFile\|MasterKey" /analysis/output/jadx_output/

# ============================================
# NATIVE LIBRARIES - PASO 9
# ============================================

# File type of native libs
file /analysis/output/native_libs/*.so

# Strings in native libs
strings /analysis/output/native_libs/libdatastore_shared_counter.so | grep -iE "(key|secret|token|password|http)"

# Symbols listing
nm -D /analysis/output/native_libs/libdatastore_shared_counter.so 2>/dev/null | head -50

# ============================================
# SEMGREP - PASO 11
# ============================================

# Run semgrep on Kotlin code
semgrep --config=p/kotlin /analysis/output/jadx_output/ --json > /analysis/output/semgrep_results.json 2>/dev/null || \
  semgrep --config=p/android /analysis/output/jadx_output/ --json > /analysis/output/semgrep_results.json 2>/dev/null || \
  echo "semgrep not available"

# ============================================
# NETWORK SECURITY - PASO 12
# ============================================

# Check network security config
cat /analysis/output/apktool_output/res/xml/network_security_config.xml 2>/dev/null || \
  echo "No network_security_config.xml found"

# ============================================
# EXTRACTION COMMANDS
# ============================================

# Extract firebase config
unzip -p /analysis/analysis/app-debug.apk res/raw/firebase_common_keep.xml
unzip -p /analysis/analysis/app-debug.apk google-services.json 2>/dev/null || echo "Not found"

# Extract AndroidManifest
unzip -p /analysis/analysis/app-debug.apk AndroidManifest.xml > /analysis/output/AndroidManifest.xml

# ============================================
# OUTPUT FILES
# ============================================

# Generated outputs in /analysis/output/
ls -la /analysis/output/

# ============================================
# DEPENDENCIES ANALYSIS
# ============================================

# Check META-INF
unzip -p /analysis/analysis/app-debug.apk META-INF/MANIFEST.MF | head -50

# Check gradle dependencies
find /analysis/output/jadx_output/ -name "build.gradle" -o -name "*.gradle" 2>/dev/null
